cmake_minimum_required(VERSION 3.20)
project(LumitPreview LANGUAGES CXX RC)
option(LUMIT_REQUIRE_GPU "Require both GPU delivery adapters for distribution" OFF)
add_library(LumitPreview SHARED preview.cpp picker.rc)
target_compile_features(LumitPreview PRIVATE cxx_std_17)
target_include_directories(LumitPreview PRIVATE ../examples/extensions/sort/openfx)
target_compile_definitions(LumitPreview PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(LumitPreview PRIVATE ws2_32 shell32 comdlg32 gdi32)
set_target_properties(LumitPreview PROPERTIES PREFIX "" SUFFIX ".ofx")
find_package(CUDAToolkit QUIET)
if(CUDAToolkit_FOUND)
    set(arch compute_75)
    if(CUDAToolkit_VERSION VERSION_LESS 13)
        set(arch compute_52)
    endif()
    set(ptx "${CMAKE_CURRENT_BINARY_DIR}/delivery.ptx")
    add_custom_command(OUTPUT "${ptx}"
        COMMAND "${CUDAToolkit_NVCC_EXECUTABLE}" --ptx -arch=${arch} -allow-unsupported-compiler
            -ccbin "${CMAKE_CXX_COMPILER}" "${CMAKE_CURRENT_SOURCE_DIR}/delivery.cu" -o "${ptx}"
        DEPENDS delivery.cu VERBATIM)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/gpu.rc" "102 RCDATA \"${ptx}\"\n")
    set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/gpu.rc" PROPERTIES OBJECT_DEPENDS "${ptx}")
    target_sources(LumitPreview PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/gpu.rc" "${ptx}")
    target_include_directories(LumitPreview PRIVATE ${CUDAToolkit_INCLUDE_DIRS})
    target_compile_definitions(LumitPreview PRIVATE LUMIT_CUDA)
    target_link_libraries(LumitPreview PRIVATE d3d11 dxgi)
endif()
find_package(OpenCL QUIET)
if(OpenCL_FOUND)
    target_include_directories(LumitPreview PRIVATE ${OpenCL_INCLUDE_DIRS})
    target_compile_definitions(LumitPreview PRIVATE LUMIT_OPENCL)
    target_link_libraries(LumitPreview PRIVATE d3d11 dxgi)
endif()
if(LUMIT_REQUIRE_GPU AND (NOT CUDAToolkit_FOUND OR NOT OpenCL_FOUND))
    message(FATAL_ERROR "Install the CUDA compiler and OpenCL development headers before packaging Resolve linking.")
endif()
